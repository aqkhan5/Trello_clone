// 1. TYPES & INTERFACES

export interface UserProfile {
  name: string;
  email: string;
  avatarUrl: string;
  status?: 'online' | 'busy' | 'away';

export interface WorkspaceItem {

  name: string;

export interface NavItem {
  id: string;

  icon: React.ComponentType<{ className?: string }>;
  badge?: string | number;
  href?: string;

export interface AppLayoutProps {
  children: React.ReactNode;
  currentUser?: UserProfile;
}

// ==========================================
// 3. COLLAPSIBLE SIDEBAR COMPONENT (`Sidebar.tsx`)
// ==========================================

export interface SidebarProps {
  isCollapsed: boolean;
  onToggleCollapse: () => void;
  activeRoute?: string;
}

export const Sidebar: React.FC<SidebarProps> = ({
  isCollapsed,
  onToggleCollapse,
  activeRoute = 'boards'
}) => {
  const [isWorkspacesExpanded, setIsWorkspacesExpanded] = useState(true);

  const mainNavItems: NavItem[] = [
    { id: 'boards', label: 'Boards', icon: Layout, badge: '12' },
    { id: 'templates', label: 'Templates', icon: FileText },
    { id: 'home', label: 'Home Feed', icon: Home }
  ];

  const workspaces: WorkspaceItem[] = [
    { id: 'ws-1', name: 'Product & Engineering', initials: 'PE', iconBg: 'bg-[#0052cc]', isPrivate: true },
    { id: 'ws-2', name: 'Marketing Campaign Q3', initials: 'MC', iconBg: 'bg-emerald-600', isPrivate: false },
    { id: 'ws-3', name: 'Design Systems', initials: 'DS', iconBg: 'bg-violet-600', isPrivate: true }
  ];

  return (
    <aside
      className={`group/sidebar relative flex flex-col border-r border-slate-200 bg-white transition-[width] duration-200 ease-in-out select-none ${
        isCollapsed ? 'w-16' : 'w-64'
      }`}
    >
      {/* Top Header / Collapse Trigger */}
      <div className="flex h-12 items-center justify-between px-3 border-b border-slate-100">
        {!isCollapsed && (
          <span className="text-xs font-semibold text-slate-400 uppercase tracking-wider">
            Workspace Hub
          </span>
        )}
        <button
          onClick={onToggleCollapse}
          title={isCollapsed ? 'Expand sidebar' : 'Collapse sidebar'}
          className={`flex h-8 w-8 items-center justify-center rounded-lg text-slate-500 hover:bg-slate-100 hover:text-slate-800 transition-colors ${
            isCollapsed ? 'mx-auto' : ''
          }`}
        >
          {isCollapsed ? (
            <PanelLeftOpen className="h-4.5 w-4.5" />
          ) : (
            <PanelLeftClose className="h-4.5 w-4.5" />
          )}
        </button>
      </div>

      {/* Main Navigation Links */}
      <div className="flex-1 overflow-y-auto px-2 py-3 space-y-1">
        {mainNavItems.map((item) => {
          const Icon = item.icon;
          const isActive = activeRoute === item.id;
          return (
            <button
              key={item.id}
              className={`flex w-full items-center gap-3 rounded-lg px-2.5 py-2 text-sm font-medium transition-all ${
                isActive
                  ? 'bg-blue-50 text-[#0052cc] font-semibold'
                  : 'text-slate-700 hover:bg-slate-100'
              } ${isCollapsed ? 'justify-center' : 'justify-between'}`}
              title={isCollapsed ? item.label : undefined}
            >
              <div className="flex items-center gap-3 min-w-0">
                <Icon className={`h-4.5 w-4.5 shrink-0 ${isActive ? 'text-[#0052cc]' : 'text-slate-500'}`} />
                {!isCollapsed && <span className="truncate">{item.label}</span>}
              </div>
              {!isCollapsed && item.badge && (
                <span className="rounded-full bg-slate-100 px-2 py-0.5 text-xs font-semibold text-slate-600">
                  {item.badge}
                </span>
              )}
            </button>
          );
        })}

        <div className="my-3 border-t border-slate-100" />

        {/* Workspaces Accordion */}
        <div className="space-y-1">
          {!isCollapsed ? (
            <div className="flex items-center justify-between px-2 py-1">
              <button
                onClick={() => setIsWorkspacesExpanded(!isWorkspacesExpanded)}
                className="flex items-center gap-1.5 text-xs font-semibold text-slate-500 hover:text-slate-800 uppercase tracking-wider"
              >
                <span>Workspaces</span>
                <ChevronDown
                  className={`h-3.5 w-3.5 transition-transform ${
                    isWorkspacesExpanded ? '' : '-rotate-90'
                  }`}
                />
              </button>
              <button
                title="Add Workspace"
                className="rounded p-1 text-slate-400 hover:bg-slate-100 hover:text-slate-700"
              >
                <Plus className="h-3.5 w-3.5" />
              </button>
            </div>
          ) : (
            <div className="flex justify-center py-1">
              <Briefcase className="h-4 w-4 text-slate-400" />
            </div>
          )}

          {(!isCollapsed ? isWorkspacesExpanded : true) && (
            <div className="space-y-0.5">
              {workspaces.map((ws) => (
                <button
                  key={ws.id}
                  className={`flex w-full items-center gap-2.5 rounded-lg px-2.5 py-1.5 text-sm text-slate-700 hover:bg-slate-100 transition-colors ${
                    isCollapsed ? 'justify-center' : ''
                  }`}
                  title={isCollapsed ? ws.name : undefined}
                >
                  <div
                    className={`flex h-6 w-6 shrink-0 items-center justify-center rounded text-[11px] font-bold text-white ${ws.iconBg}`}
                  >
                    {ws.initials}
                  </div>
                  {!isCollapsed && (
                    <div className="flex flex-1 items-center justify-between truncate">
                      <span className="truncate text-xs font-medium text-slate-800">{ws.name}</span>
                      {ws.isPrivate && <Lock className="h-3 w-3 text-slate-400 shrink-0 ml-1" />}
                    </div>
                  )}
                </button>
              ))}
            </div>
          )}
        </div>
      </div>

      {/* Bottom Utility Links */}
      <div className="border-t border-slate-200 p-2 space-y-0.5 bg-slate-50/50">
        <button
          className={`flex w-full items-center gap-3 rounded-lg px-2.5 py-2 text-xs font-medium text-slate-600 hover:bg-slate-100 hover:text-slate-900 transition-colors ${
            isCollapsed ? 'justify-center' : ''
          }`}
          title={isCollapsed ? 'Members & Access' : undefined}
        >
          <Users className="h-4 w-4 text-slate-500 shrink-0" />
          {!isCollapsed && <span>Members & Access</span>}
        </button>

        <button
          className={`flex w-full items-center gap-3 rounded-lg px-2.5 py-2 text-xs font-medium text-slate-600 hover:bg-slate-100 hover:text-slate-900 transition-colors ${
            isCollapsed ? 'justify-center' : ''
          }`}
          title={isCollapsed ? 'Workspace Settings' : undefined}
        >
          <Settings className="h-4 w-4 text-slate-500 shrink-0" />
          {!isCollapsed && <span>Workspace Settings</span>}
        </button>
      </div>
    </aside>
  );
};

  children,
  currentUser = {

    avatarUrl: 'https://images.unsplash.com/photo-1494790108377-be9c29b29330?w=150&auto=format&fit=crop',
    status: 'online'
  }
}) => {
  const [isSidebarCollapsed, setIsSidebarCollapsed] = useState(false);
  const [isMobileMenuOpen, setIsMobileMenuOpen] = useState(false);

  return (
    <div className="flex h-screen w-full flex-col bg-[#faf9ff] text-slate-900 font-sans antialiased overflow-hidden">
      {/* 1. Persistent Top Navbar */}

        currentUser={currentUser}

      <div className="relative flex flex-1 overflow-hidden">

        <div className="hidden md:flex h-full shrink-0">
          <Sidebar
            isCollapsed={isSidebarCollapsed}
            onToggleCollapse={() => setIsSidebarCollapsed(!isSidebarCollapsed)}
            activeRoute="boards"
          />
        </div>

        {/* Mobile Sliding Drawer / Sheet */}
        {isMobileMenuOpen && (
          <div className="fixed inset-0 z-50 flex md:hidden">
            {/* Backdrop */}
            <div
              className="fixed inset-0 bg-slate-900/40 backdrop-blur-xs transition-opacity"
              onClick={() => setIsMobileMenuOpen(false)}
            />
            {/* Drawer Content */}
            <div className="relative flex w-72 max-w-xs flex-col bg-white shadow-2xl z-10 animate-in slide-in-from-left duration-200">

                  <div className="flex h-7 w-7 items-center justify-center rounded-md bg-[#0052cc] text-white">
                    <Trello className="h-4 w-4" />
                  </div>
                  <span className="font-bold text-slate-900">Trello Navigation</span>
                </div>
                <button
                  onClick={() => setIsMobileMenuOpen(false)}
                  className="rounded-lg p-1.5 text-slate-500 hover:bg-slate-100"
                >
                  <X className="h-5 w-5" />
                </button>
              </div>
              <div className="flex-1 overflow-y-auto">
                <Sidebar
                  isCollapsed={false}
                  onToggleCollapse={() => setIsMobileMenuOpen(false)}
                  activeRoute="boards"
                />
              </div>
            </div>
          </div>
        )}

        {/* 3. Flexible Main Content Viewport */}
        <main className="flex-1 overflow-y-auto bg-[#faf9ff] p-4 sm:p-6 lg:p-8">
          <div className="mx-auto max-w-7xl">
            {children}
          </div>
        </main>
      </div>
    </div>
  );
};

export default AppLayout;
